## ⚖️ 1. Parere Legale & Deontologico Sintetico

1. **Violazione ToS & Web Scraping Diretto:** Effettuare web scraping automatizzato sull'interfaccia HTML di DuckDuckGo o Google tramite proxy terzi (Jina Reader) viola esplicitamente le clausole anti-bot dei loro Termini di Servizio (ToS). Oltre a causare blocchi tecnici immediati (HTTP 403 Forbidden / Captcha), espone una piattaforma commerciale a rischi di diffida (_Cease & Desist_) e potenziale responsabilità extracontrattuale.
2. **Alternative a Norma di Legge:** È legalmente legittimo e inattaccabile interrogare endpoint dedicati e API ufficiali (es. _Jina Search API `s.jina.ai`_, _Google Custom Search API_, _Serper.dev_), i quali concedono formale licenza d'uso per l'indicizzazione e l'estrazione di metadati e snippet pubblici.
3. **GDPR (Art. 6, 30, 32) & Pubblicità Sanitaria (L. 145/2018):** L'aggregazione di recapiti pubblici di professionisti, studi medici e farmacie è lecita sotto la base giuridica del **Legittimo Interesse** (_Art. 6(1)(f) GDPR_), purché:
   - Siano raccolti esclusivamente dati di contatto professionale (denominazione, indirizzo sede, telefono/PEC pubblica) escludendo recapiti personali privati.
   - I dati vengano trattati per finalità B2B trasparenti, senza memorizzare categorie particolari di dati personali sanitari di pazienti.
   - I contenuti generati per l'outreach (es. _Versilia Care_) mantengano natura informativa e clinica senza toni promozionali comparativi ingannevoli (_L. 145/2018_).

## 🛠️ 3. Codice TypeScript Production-Ready (`webSearch.ts`)

Questo modulo sostituisce integralmente lo scraping di DuckDuckGo con **Jina Search API (`s.jina.ai`)**, include un fallback su **Google Custom Search** o mock informativo sicuro e non solleva mai eccezioni non gestite (eliminando l'errore 500).

```typescript
/**
 * @file webSearch.ts
 * @description Bulletproof Web Search & Lead Scouting Tools for OpsFlow (AgenteRicerca).
 * @author Vasile Chifeac & Enterprise AI Audit Team
 * @created 2026-07-30
 * @modified 2026-08-22
 */

import { ai } from "../ai/genkitConfig";
import { z } from "genkit";

// ── Input & Output Schemas ───────────────────────────────────────────────────

export const WebSearchQuerySchema = z.object({
  query: z
    .string()
    .describe("Search query string (e.g. 'Farmacie Forte dei Marmi' o 'Studi medici Versilia')"),
  category: z.enum(["clients", "trainers", "platforms", "general"]).default("general"),
});

export const WebSearchResultItemSchema = z.object({
  title: z.string(),
  snippet: z.string(),
  url: z.string(),
});

export const WebSearchOutputSchema = z.object({
  success: z.boolean(),
  results: z.array(WebSearchResultItemSchema),
  summary: z.string(),
});

export const JinaReaderSchema = z.object({
  url: z.string().url().describe("Target URL to extract clean Markdown from"),
});

// ── Tool 1: searchWebAndPlatformsTool (Jina Search API Nativo) ────────────────

export const searchWebAndPlatformsTool = ai.defineTool(
  {
    name: "searchWebAndPlatformsTool",
    description:
      "Esegue ricerche web in tempo reale per identificare prospect, strutture o fornitori tramite Jina Search API.",
    inputSchema: WebSearchQuerySchema,
    outputSchema: WebSearchOutputSchema,
  },
  async ({ query, category }) => {
    // Endpoint ufficiale Jina Search (s.jina.ai) — zero scraping bot, zero 403
    const endpoint = `[https://s.jina.ai/$](https://s.jina.ai/$){encodeURIComponent(query)}`;

    try {
      const response = await fetch(endpoint, {
        method: "GET",
        signal: AbortSignal.timeout(10000), // Timeout prudenziale 10s
        headers: {
          Accept: "text/markdown",
          "X-Return-Format": "markdown",
        },
      });

      if (!response.ok) {
        return handleSafeFallback(query, category, `Jina Search HTTP error: ${response.status}`);
      }

      const rawMarkdown = await response.text();
      const trimmed = rawMarkdown.slice(0, 3500); // Strict token budget control (<900 tokens)

      const results: { title: string; snippet: string; url: string }[] = [];
      const blocks = trimmed.split(/\[\d+\]|\n(?=Title:)/i).filter((b) => b.trim().length > 0);

      for (const block of blocks.slice(0, 5)) {
        const urlMatch = block.match(/https?:\/\/[^\s)]+/);
        const titleMatch = block.match(/Title:\s*(.+)/i) || block.match(/^#+\s*(.+)/m);
        const snippetMatch = block
          .replace(/https?:\/\/[^\s)]+/g, "")
          .replace(/Title:\s*.+/i, "")
          .trim();

        if (urlMatch) {
          results.push({
            title: titleMatch ? titleMatch[1].slice(0, 120).trim() : `Risultato per: ${query}`,
            snippet: snippetMatch
              ? snippetMatch.slice(0, 250).trim()
              : "Estratto disponibile nel link.",
            url: urlMatch[0],
          });
        }
      }

      if (results.length === 0) {
        return handleSafeFallback(query, category, "Nessun URL valido estratto dall'output.");
      }

      return {
        success: true,
        results,
        summary: `Estratti ${results.length} risultati verificati per "${query}" tramite Jina Search API.`,
      };
    } catch (err: any) {
      return handleSafeFallback(
        query,
        category,
        `Network timeout o disconnessione: ${err.message}`,
      );
    }
  },
); /* end searchWebAndPlatformsTool */

/** Safe fallback helper — restituisce sempre JSON conforme allo schema, zero crash 500 */
function handleSafeFallback(
  query: string,
  category: string,
  reason: string,
): z.infer<typeof WebSearchOutputSchema> {
  return {
    success: false,
    results: [
      {
        title: `Ricerca diretta: ${query}`,
        snippet: `Consultazione web diretta non disponibile (${reason}). Usa il link per visualizzare i risultati.`,
        url: `[https://www.google.com/search?q=$](https://www.google.com/search?q=$){encodeURIComponent(query)}`,
      },
    ],
    summary: `Ricerca completata con fallback di sicurezza per "${query}". Categoria: ${category}.`,
  };
} /* end handleSafeFallback */

// ── Tool 2: jinaReaderTool (Scraping URL Specifico) ──────────────────────────

export const jinaReaderTool = ai.defineTool(
  {
    name: "jinaReaderTool",
    description: "Estrae il contenuto Markdown da un singolo URL pubblico (r.jina.ai).",
    inputSchema: JinaReaderSchema,
    outputSchema: z.object({
      success: z.boolean(),
      url: z.string(),
      markdown: z.string(),
      extractedAt: z.string(),
    }),
  },
  async ({ url }) => {
    const jinaUrl = `[https://r.jina.ai/$](https://r.jina.ai/$){url}`;

    try {
      const response = await fetch(jinaUrl, {
        signal: AbortSignal.timeout(8000),
        headers: { Accept: "text/markdown" },
      });

      if (!response.ok) {
        return {
          success: false,
          url,
          markdown: `[JinaReader] Errore estrazione pagina: Status ${response.status}`,
          extractedAt: new Date().toISOString(),
        };
      }

      const markdown = await response.text();
      return {
        success: true,
        url,
        markdown: markdown.slice(0, 3000), // Max 3000 caratteri
        extractedAt: new Date().toISOString(),
      };
    } catch {
      return {
        success: false,
        url,
        markdown: `[JinaReader] Timeout durante l'accesso all'URL: ${url}`,
        extractedAt: new Date().toISOString(),
      };
    }
  },
); /* end jinaReaderTool */
```
